export default { async fetch(request, env) { const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", "Content-Type": "application/json" }; const json = (data, status = 200) => new Response(JSON.stringify(data), { status, headers: corsHeaders }); if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: corsHeaders }); } if (request.method !== "POST") { return json({ error: "Use POST to chat with MediSchool AI." }, 405); } if (!env.GROQ_API_KEY) { return json({ error: "GROQ_API_KEY secret is not configured." }, 500); } try { const body = await request.json(); if (!Array.isArray(body.messages) || body.messages.length === 0) { return json({ error: "Please send a message." }, 400); } const messages = body.messages.slice(-16); const allowed = messages.every(m => m && ["user", "assistant"].includes(m.role) && typeof m.content === "string" && m.content.length <= 4000 ); if (!allowed) { return json({ error: "Invalid message format." }, 400); } const systemPrompt = ` You are MediSchool AI, a school health education assistant. Help students, teachers and parents with general health education, hygiene, nutrition, sleep, wellbeing and basic first-aid information. Rules: - Never claim to diagnose a disease or replace a doctor. - Use clear, kind, age-appropriate language. - Answer in the language the user uses, including Hindi or Hinglish. - Do not prescribe medicines or dosages to children. - Encourage a parent/guardian, school nurse or healthcare professional for symptoms, injuries or medical decisions. - For serious symptoms such as difficulty breathing, unconsciousness, severe bleeding, seizures or a severe allergic reaction, advise immediate adult help and local emergency services. - For self-harm or immediate danger, encourage immediate trusted-adult support and emergency help. - Ask users not to share names, addresses or other private information. - Give cautious, general first-aid advice only. - Be concise and easy to understand. - This chatbot is educational and is not an emergency service. `; const upstream = await fetch( "https://api.groq.com/openai/v1/chat/completions", { method: "POST", headers: { "Authorization": `Bearer ${env.GROQ_API_KEY}`, "Content-Type": "application/json" }, body: JSON.stringify({ model: "llama-3.3-70b-versatile", messages: [ { role: "system", content: systemPrompt }, ...messages ], temperature: 0.3, max_tokens: 700 }) } ); if (!upstream.ok) { const details = await upstream.text(); console.error("Groq API error:", upstream.status, details); return json({ error: upstream.status === 429 ? "AI usage limit reached. Please try again later." : "The AI provider request failed. Check Worker logs and Groq settings." }, 502); } const result = await upstream.json(); const reply = result.choices?.[0]?.message?.content; if (!reply) { return json({ error: "No answer was returned by the AI." }, 502); } return json({ reply }); } catch (error) { console.error("Worker error:", error); return json({ error: "Invalid request or server error." }, 500); } } };
